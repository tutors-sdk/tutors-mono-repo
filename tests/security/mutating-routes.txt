# Every route that changes state, and who may call it (runway tier M; the seed
# of tier F's authorisation matrix). tests/security/security.test.ts fails when
# a +server.ts exports POST/PUT/PATCH/DELETE or a +page.server.ts declares an
# action that is not listed here, or when a listed route disappears.
# The container smoke test posts a cross-site form to each route and expects 403.
#
# Format: `<app> <METHOD> <route id> | who: <callers>`
# Add `| hook` for routes a hook mounts (discovery cannot see them).

time POST /api/sync | who: anyone who can reach the time app today; the handler has no authentication or role check (tier F must add one)
reader POST /auth/signin/github | hook | who: anyone; Auth.js starts GitHub OAuth, SvelteKit checks cross-site form origins
reader POST /auth/signout | hook | who: anyone clearing their browser cookies; SvelteKit checks cross-site form origins
reader POST /api/auth/sign-in/social | hook | who: anyone; GitHub only, Better Auth verifies origin and return URL
reader POST /api/auth/callback/github | hook | who: OAuth callback holder; Better Auth verifies signed state (GitHub normally uses GET)
reader POST /api/auth/sign-out | hook | who: session holder; Better Auth verifies origin and deletes browser cookies
reader POST /api/auth/get-session | hook | who: session holder; Better Auth verifies cookies and renews the session
reader POST /api/auth/update-user | hook | who: session holder; Better Auth verifies origin, adapter rejects githubId/login changes including null
